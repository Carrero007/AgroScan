-- Criação do Banco de Dados
CREATE DATABASE AgroScan;
GO

USE AgroScan;
GO

-- 1. Tabela de Usuários (com GeminiApiKeyEnc)
CREATE TABLE [dbo].[Usuarios] (
    [UsuarioId]         INT             IDENTITY (1, 1) NOT NULL,
    [CPF]               VARCHAR (11)    NOT NULL,
    [Nome]              VARCHAR (100)   DEFAULT ('') NOT NULL,
    [SenhaHash]         VARCHAR (256)   NOT NULL,
    [Cep]               VARCHAR (8)     NOT NULL,
    [Cidade]            VARCHAR (100)   NOT NULL,
    [Estado]            CHAR (2)        NOT NULL,
    [Ativo]             BIT             DEFAULT ((1)) NOT NULL,
    [DataCriacao]       DATETIME        DEFAULT (getdate()) NOT NULL,
    [UltimoLogin]       DATETIME        NULL,
    [GeminiApiKeyEnc]   VARCHAR (512)   NULL,
    CONSTRAINT [PK_Usuarios] PRIMARY KEY CLUSTERED ([UsuarioId] ASC),
    CONSTRAINT [UQ_Usuarios_CPF] UNIQUE NONCLUSTERED ([CPF] ASC)
);
GO

-- 2. Tabela de Hortaliças
CREATE TABLE [dbo].[Hortalicas] (
    [HortalicaId]          INT             IDENTITY (1, 1) NOT NULL,
    [NomeCientifico]       VARCHAR (255)   NOT NULL,
    [NomePopular]          VARCHAR (255)   NULL,
    [Familia]              VARCHAR (100)   NULL,
    [Categoria]            VARCHAR (50)    NULL,
    [CicloVida]            VARCHAR (50)    NULL,
    [DiasGerminacao]       INT             NULL,
    [DiasColheita]         INT             NULL,
    [Espacamento]          VARCHAR (100)   NULL,
    [ProfundidadeSemeio]   VARCHAR (50)    NULL,
    [Clima]                VARCHAR (100)   NULL,
    [TemperaturaMin]       DECIMAL (4, 1)  NULL,
    [TemperaturaMax]       DECIMAL (4, 1)  NULL,
    [Luminosidade]         VARCHAR (100)   NULL,
    [Irrigacao]            VARCHAR (100)   NULL,
    [NecessidadeAgua]      VARCHAR (50)    NULL,
    [TipoSolo]             VARCHAR (255)   NULL,
    [PHMin]                DECIMAL (3, 1)  NULL,
    [PHMax]                DECIMAL (3, 1)  NULL,
    [Adubacao]             VARCHAR (MAX)   NULL,
    [PragasPrincipais]     VARCHAR (MAX)   NULL,
    [DoencasPrincipais]    VARCHAR (MAX)   NULL,
    [Origem]               VARCHAR (255)   NULL,
    [ValorNutricional]     VARCHAR (MAX)   NULL,
    [Observacoes]          VARCHAR (MAX)   NULL,
    [DataCriacao]          DATETIME        DEFAULT (getdate()) NULL,
    [DataAtualizacao]      DATETIME        NULL,
    CONSTRAINT [PK_Hortalicas] PRIMARY KEY CLUSTERED ([HortalicaId] ASC)
);
GO

-- 3. Tabela de Diagnósticos (Depende de Usuarios e Hortalicas)
CREATE TABLE [dbo].[Diagnosticos] (
    [DiagnosticoId]         INT             IDENTITY (1, 1) NOT NULL,
    [UsuarioId]             INT             NULL,
    [HortalicaId]           INT             NULL,
    [TipoDiagnostico]       NVARCHAR (80)   NULL,
    [NomeDoenca]            NVARCHAR (200)  NULL,
    [NomeCientifico]        NVARCHAR (200)  NULL,
    [AgenteCausador]        NVARCHAR (200)  NULL,
    [Confianca]             INT             NULL,
    [GravidadeNivel]        INT             NULL,
    [Gravidade]             VARCHAR (20)    NULL,
    [SintomasObservados]    NVARCHAR (MAX)  NULL,
    [Tratamento]            NVARCHAR (MAX)  NULL,
    [TratamentoEcologico]   NVARCHAR (MAX)  NULL,
    [TratamentoQuimico]     NVARCHAR (MAX)  NULL,
    [Prevencao]             NVARCHAR (MAX)  NULL,
    [RiscoPropagacao]       VARCHAR (20)    NULL,
    [RiscoPropagacaoNivel]  INT             NULL,
    [PlantasAfetadas]       NVARCHAR (MAX)  NULL,
    [CondicoesFavoraveis]   NVARCHAR (MAX)  NULL,
    [ImagemBase64]          NVARCHAR (MAX)  NULL,
    [Latitude]              FLOAT (53)      NULL,
    [Longitude]             FLOAT (53)      NULL,
    [DataDiagnostico]       DATETIME        DEFAULT (getdate()) NULL,
    [StatusTratamento]      NVARCHAR (20)   DEFAULT ('nao iniciado') NOT NULL,
    [DataInicioTratamento]  DATETIME        NULL,
    [DataConclusaoTratamento] DATETIME      NULL,
    [TratamentoPassosJson]  NVARCHAR (MAX)  NULL,
    CONSTRAINT [PK_Diagnosticos] PRIMARY KEY CLUSTERED ([DiagnosticoId] ASC),
    CONSTRAINT [FK_Diag_Hortalica] FOREIGN KEY ([HortalicaId]) REFERENCES [dbo].[Hortalicas] ([HortalicaId])
);
GO

-- Índices para Diagnosticos
CREATE NONCLUSTERED INDEX [IX_Diagnosticos_UsuarioId]
ON [dbo].[Diagnosticos] ([UsuarioId] ASC);
GO

CREATE NONCLUSTERED INDEX [IX_Diagnosticos_Data]
ON [dbo].[Diagnosticos] ([DataDiagnostico] DESC);
GO

-- 4. Tabela de TratamentoEtapas (Depende de Diagnosticos)
CREATE TABLE [dbo].[TratamentoEtapas] (
    [EtapaId]        INT             IDENTITY (1, 1) NOT NULL,
    [DiagnosticoId]  INT             NOT NULL,
    [Descricao]      NVARCHAR (500)  NOT NULL,
    [Ordem]          INT             NOT NULL,
    [Concluida]      BIT             DEFAULT ((0)) NOT NULL,
    [DataConclusao]  DATETIME        NULL,
    CONSTRAINT [PK_TratamentoEtapas] PRIMARY KEY CLUSTERED ([EtapaId] ASC),
    CONSTRAINT [FK_TratamentoEtapas_Diagnostico] FOREIGN KEY ([DiagnosticoId]) REFERENCES [dbo].[Diagnosticos] ([DiagnosticoId]) ON DELETE CASCADE
);
GO

CREATE NONCLUSTERED INDEX [IX_TratamentoEtapas_DiagnosticoId]
ON [dbo].[TratamentoEtapas] ([DiagnosticoId] ASC);
GO

-- 5. Tabela de AuditLog
CREATE TABLE [dbo].[AuditLog] (
    [LogId]        INT             IDENTITY (1, 1) NOT NULL,
    [UsuarioId]    INT             NULL,
    [Acao]         VARCHAR (100)   NOT NULL,
    [IP]           VARCHAR (45)    NULL,
    [Modelo]       VARCHAR (100)   NULL,
    [TokensUsados] INT             NULL,
    [CriadoEm]     DATETIME        DEFAULT (getdate()) NOT NULL,
    CONSTRAINT [PK_AuditLog] PRIMARY KEY CLUSTERED ([LogId] ASC)
);
GO

-- 6. Tabela de RefreshTokens (Depende de Usuarios)
CREATE TABLE [dbo].[RefreshTokens] (
    [Id]         INT             IDENTITY (1, 1) NOT NULL,
    [UsuarioId]  INT             NOT NULL,
    [Token]      VARCHAR (512)   NOT NULL,
    [Expiracao]  DATETIME        NOT NULL,
    [Revogado]   BIT             DEFAULT ((0)) NOT NULL,
    [CriadoEm]   DATETIME        DEFAULT (getdate()) NOT NULL,
    [IP]         VARCHAR (45)    NULL,
    CONSTRAINT [PK_RefreshTokens] PRIMARY KEY CLUSTERED ([Id] ASC),
    CONSTRAINT [FK_RefreshTokens_Usuarios] FOREIGN KEY ([UsuarioId]) REFERENCES [dbo].[Usuarios] ([UsuarioId]) ON DELETE CASCADE
);
GO
